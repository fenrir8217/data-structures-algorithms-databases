\subsubsection{Q3. Time Complexity of Algorithms}

\textbf{1 Multiply all elements in the array}

\begin{lstlisting}
def product(arr):
    n = len(arr)
    x = 1
    i = 0
    while i < n:
        x *= arr[i]
        i += 1
    return x
\end{lstlisting}

\textbf{Answer:} $O(n)$

\textbf{2 Modify the last value}

\begin{lstlisting}
def modify(arr):
    if len(arr) == 0:
        raise Exception("Array is empty")

    last = arr[-1]
    if last < 0:
        last = -last

    arr[-1] = last
\end{lstlisting}

\textbf{Answer:} $O(1)$

\textbf{3 Find the largest (method 1, brute force)}

\begin{lstlisting}
def largest1(arr):
    n = len(arr)
    max_val = 0

    for i in range(n):
        is_largest = True
        for j in range(n):
            if arr[i] < arr[j]:
                is_largest = False
        if is_largest:
            max_val = arr[i]

    return max_val
\end{lstlisting}

\textbf{Answer:} $O(n^2)$

\textbf{4 Find the largest (method 2, single scan)}

\begin{lstlisting}
def largest2(arr):
    n = len(arr)
    max_val = 0

    if n == 0:
        return 0
    else:
        max_val = arr[0]
        for i in range(n):
            if arr[i] > max_val:
                max_val = arr[i]

    return max_val
\end{lstlisting}

\textbf{Answer:} $O(n)$

\textbf{5 Find the largest (method 3, via sort)}

\begin{lstlisting}
def largest3(arr):
    arr.sort()  # assume O(n log n)

    if len(arr) == 0:
        return 0
    else:
        last = arr[-1]
        return last
\end{lstlisting}

\textbf{Answer:} $O(n \log n)$
